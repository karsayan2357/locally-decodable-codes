\chapter*{Notations}

\begin{tabular}{ll}
    $\mathbb{F}_q$ & Finite field of size $q$ \\[0.3em]
$\mathbb{F}_q^*$ & Multiplicative group of $\mathbb{F}_q$ \\ [0.3em]
$\mathbb{N}$ & Set of natural numbers \\[0.3em]
$[n]$ & The set $\{1,2,\dots,n\}$ \\[0.3em]
$\emptyset$ & Empty set \\[0.3em]
$X \sim \mathscr{D}$ & Random Variable $X$ having distribution $\mathscr{D}$ \\[0.3em]
$\Delta(\cdot,\cdot)$ & Relative Hamming Distance \\[0.3em]
$\mathbb{E}[X]$ & Expectation of random variable $X$ \\[0.3em]
$\mathbf{w}(l)$  & $l$-th coordinate of $\mathbf{w}\in\mathbb{F}_q^n$ \\ [0.3em]
$D$-evaluation $h$ defined over a domain $D$ & vector of values of $h$ at all points of $D$ \\[0.3em]
$\mathrm{Unif}(S)$ & Uniform distribution over the set $S$ \\[0.3em]
$q$ & a prime power, if not mentioned otherwise \\[0.3em]
$g^{\bfw}$ & $(g^{\bfw(1)}, \ldots, g^{\bfw(n)})$ \\[0.3em]
$M_{\mathbf{w},\mathbf{v}}$ & $\{g^{\mathbf{w}+\lambda\mathbf{v}} \mid \lambda \in \mathbb{Z}_m\}$ \\[0.3em]
$\mathrm{mon}_{\mathbf{u}}(z_1,\ldots,z_n)$ & $\prod_{\ell} z_\ell^{\mathbf{u}(\ell)}$ in $\mathbb{F}_q[z_1,\dots,z_n]/(z_1^m=1,\dots,z_n^m=1)$ \\[0.3em]
$(\bfu,\bfv)$ & inner product of vectors $\bfu$ and $\bfv$ \\[0.3em]
$\mathrm{supp}(h(y))$ & $\{e\in\bbn_{\geq0} : y^e\text{ has positive coefficient in }h\}$ \\[0.3em]
$b$-bounded set & every element of the set($\subseteq\bbr$) is less that $b$\\[0.3em]
$\mathrm{agr}(e,M)$ & number of agreements of $M$-eval of $F$ and $C_m$-eval of $h$\\[0.3em]
$\mathrm{weight}(e,M)$ & $\max\{\mathrm{agr}(e,M) : h\in\bbf[y], h(0) = e, \mathrm{supp}(h)\subseteq S\cup\{0\}\}$
\end{tabular}
