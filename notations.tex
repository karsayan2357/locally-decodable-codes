\chapter*{Notations}

\begin{tabular}{ll}
	$\mathbb{F}_q$ & Finite field of size $q$ \\[0.3em]
	$\mathbb{F}_q^*$ & Multiplicative group of $\mathbb{F}_q$ \\ [0.3em]
	$\mathbb{N}$ & Set of natural numbers \\[0.3em]
	$[n]$ & The set $\{1,2,\dots,n\}$ \\[0.3em]
	$\emptyset$ & Empty set \\[0.3em]
	$X \sim \mathscr{D}$ & Random Variable $X$ having distribution $\mathscr{D}$ \\[0.3em]
	$\Delta(\cdot,\cdot)$ & Relative Hamming Distance \\[0.3em]
	$\mathbb{E}[X]$ & Expectation of random variable $X$ \\[0.3em]
	$\mathbf{w}(l)$  & $l$-th coordinate of $\mathbf{w}\in\mathbb{F}_q^n$ \\ [0.3em]
	$D$-evaluation $h$ defined over a domain $D$ & vector of values of $h$ at all points of $D$ \\[0.3em]
	$\mathrm{Unif}(S)$ & Uniform distribution over the set $S$ \\[0.3em]
\end{tabular}
